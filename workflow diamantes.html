datos_transf <- datos |> mutate(price10 = log10(price))
datos_split <- initial_split(datos_transf, 
                                 prop = 0.8, 
                                 strata = price
)
datos_train <- training(datos_split)
datos_test  <- testing(datos_split)
datos_reci <- recipe(price10 ~ carat + cut +  color + clarity, data = datos_train)   |> 
  step_log(carat, base = 10) |> 
  step_normalize(all_numeric_predictors()) |>
  step_dummy(all_nominal_predictors()) 
tidy(datos_reci)
